--------------------------------------------------------
-- Regla de negocio: UNA cosecha activa por lote
-- Índice único condicional (concurrencia)
--------------------------------------------------------
CREATE UNIQUE INDEX UNQ_HARVEST_ONE_ACTIVE
ON HARVEST (
  CASE WHEN STATUS_ID = 1 AND ENDDATE IS NULL THEN IDPLOT END
);

--------------------------------------------------------
-- TABLA AUXILIAR: correlativo por lote
--------------------------------------------------------
CREATE TABLE PLOT_HARVEST_SEQ (
    IDPLOT   NUMBER(10,0) NOT NULL,
    NEXT_ID  NUMBER(10,0) NOT NULL,
    CONSTRAINT PLOT_HARVEST_SEQ_PK PRIMARY KEY (IDPLOT),
    CONSTRAINT PLOT_HARVEST_SEQ_FK FOREIGN KEY (IDPLOT)
        REFERENCES PLOT (IDPLOT)
);

--------------------------------------------------------
-- TRIGGER 1: IDHARVEST incremental por IDPLOT
--------------------------------------------------------
CREATE OR REPLACE TRIGGER HARVEST_BI_SET_ID
BEFORE INSERT ON HARVEST
FOR EACH ROW
DECLARE
    v_next NUMBER;
BEGIN
    IF :NEW.IDHARVEST IS NULL THEN
        BEGIN
            SELECT NEXT_ID INTO v_next
              FROM PLOT_HARVEST_SEQ
             WHERE IDPLOT = :NEW.IDPLOT
             FOR UPDATE;

            :NEW.IDHARVEST := v_next;
            UPDATE PLOT_HARVEST_SEQ
               SET NEXT_ID = v_next + 1
             WHERE IDPLOT = :NEW.IDPLOT;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                :NEW.IDHARVEST := 1;
                INSERT INTO PLOT_HARVEST_SEQ (IDPLOT, NEXT_ID)
                VALUES (:NEW.IDPLOT, 2);
        END;
    END IF;
END;
/
SHOW ERRORS TRIGGER HARVEST_BI_SET_ID;

--------------------------------------------------------
-- TRIGGER 2: Derivar estado y validar 1 activa por lote
--------------------------------------------------------
CREATE OR REPLACE TRIGGER HARVEST_BI_SET_STATUS_AND_VALIDATE
BEFORE INSERT OR UPDATE OF ENDDATE, STATUS_ID ON HARVEST
FOR EACH ROW
DECLARE
    v_exists NUMBER;
BEGIN
    -- Derivar estado/fechas 
    IF INSERTING THEN
        IF :NEW.ENDDATE IS NOT NULL THEN
            :NEW.STATUS_ID := 2;
        ELSIF :NEW.STATUS_ID IS NULL THEN
            :NEW.STATUS_ID := 1;
        END IF;
    END IF;

    IF :NEW.ENDDATE IS NOT NULL THEN
        :NEW.STATUS_ID := 2;
    ELSIF :NEW.STATUS_ID = 2 AND :NEW.ENDDATE IS NULL THEN
        :NEW.ENDDATE := TRUNC(SYSDATE);
    END IF;

    -- Validación: SOLO una activa por lote (después de fijar estado)
    IF (:NEW.STATUS_ID = 1 AND :NEW.ENDDATE IS NULL) THEN
        IF INSERTING THEN
            SELECT COUNT(*) INTO v_exists
              FROM HARVEST H
             WHERE H.IDPLOT = :NEW.IDPLOT
               AND H.STATUS_ID = 1
               AND H.ENDDATE IS NULL;
        ELSE -- UPDATING
            SELECT COUNT(*) INTO v_exists
              FROM HARVEST H
             WHERE H.IDPLOT = :NEW.IDPLOT
               AND H.STATUS_ID = 1
               AND H.ENDDATE IS NULL
               AND H.IDHARVEST <> :NEW.IDHARVEST;
        END IF;

        IF v_exists > 0 THEN
            RAISE_APPLICATION_ERROR(-20051,
                'Solo se permite una cosecha activa por lote. IDPLOT='||:NEW.IDPLOT);
        END IF;
    END IF;
END;
/
SHOW ERRORS TRIGGER HARVEST_BI_SET_STATUS_AND_VALIDATE;
