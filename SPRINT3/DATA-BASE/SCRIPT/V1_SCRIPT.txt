--------------------------------------------------------
-- TABLA DE ASOCIACIÓN: HARVEST_COLLECTOR
--------------------------------------------------------
CREATE TABLE HARVEST_COLLECTOR (
    IDPLOT        NUMBER(10,0) NOT NULL,
    IDHARVEST     NUMBER(10,0) NOT NULL,
    WORKER_CODE   VARCHAR2(6)  NOT NULL,
    ASSIGN_DATE   DATE DEFAULT SYSDATE NOT NULL,
    STATUS_ID     NUMBER       DEFAULT 1 NOT NULL  -- 1=ACTIVO, 2=INACTIVO
);

ALTER TABLE HARVEST_COLLECTOR
  ADD CONSTRAINT HARVEST_COLLECTOR_PK
  PRIMARY KEY (IDPLOT, IDHARVEST, WORKER_CODE);

-- Relación hacia la cosecha (HARVEST)
ALTER TABLE HARVEST_COLLECTOR
  ADD CONSTRAINT HARVEST_COLLECTOR_FK_HARVEST
  FOREIGN KEY (IDPLOT, IDHARVEST)
  REFERENCES HARVEST (IDPLOT, IDHARVEST)
  ENABLE;

-- Relación hacia el recolector (COLLECTOR)
ALTER TABLE HARVEST_COLLECTOR
  ADD CONSTRAINT HARVEST_COLLECTOR_FK_COLLECTOR
  FOREIGN KEY (WORKER_CODE)
  REFERENCES COLLECTOR (WORKER_CODE)
  ENABLE;

-- Check para estatus
ALTER TABLE HARVEST_COLLECTOR
  ADD CONSTRAINT HARVEST_COLLECTOR_CHK_STATUS
  CHECK (STATUS_ID IN (1, 2))
  ENABLE;
