    ENDDATE       DATE,
    PRICEPERKILO  NUMBER(10,2)  NOT NULL,

);

ALTER TABLE HARVEST
  ADD CONSTRAINT HARVEST_PK PRIMARY KEY (IDPLOT, IDHARVEST) ENABLE;

  ADD CONSTRAINT HARVEST_FK_PLOT FOREIGN KEY (IDPLOT)
      REFERENCES PLOT (IDPLOT) ENABLE;

  ADD CONSTRAINT HARVEST_FK_STATUS FOREIGN KEY (STATUS_ID)
      REFERENCES HARVEST_STATUS_DOMAIN (STATUS_ID) ENABLE;

ALTER TABLE HARVEST
  ADD CONSTRAINT HARVEST_CHK_PRICE CHECK (PRICEPERKILO >= 0) ENABLE;

ALTER TABLE HARVEST
  ADD CONSTRAINT HARVEST_CHK_STATE_DATE CHECK (
      (STATUS_ID = 1 AND ENDDATE IS NULL) OR
      (STATUS_ID = 2 AND ENDDATE IS NOT NULL)
) ENABLE;

--------------------------------------------------------
-- TABLA AUXILIAR: correlativo por lote
--------------------------------------------------------
CREATE TABLE PLOT_HARVEST_SEQ (
    IDPLOT   NUMBER(10,0) NOT NULL,
    NEXT_ID  NUMBER(10,0) NOT NULL,
    CONSTRAINT PLOT_HARVEST_SEQ_PK PRIMARY KEY (IDPLOT),
    CONSTRAINT PLOT_HARVEST_SEQ_FK FOREIGN KEY (IDPLOT)
        REFERENCES PLOT (IDPLOT)
);

--------------------------------------------------------
-- TRIGGER 1: IDHARVEST incremental por IDPLOT
--------------------------------------------------------
CREATE OR REPLACE TRIGGER HARVEST_BI_SET_ID
BEFORE INSERT ON HARVEST
FOR EACH ROW
DECLARE
    v_next NUMBER;
BEGIN
    IF :NEW.IDHARVEST IS NULL THEN
        BEGIN
            SELECT NEXT_ID INTO v_next
              FROM PLOT_HARVEST_SEQ
             WHERE IDPLOT = :NEW.IDPLOT
             FOR UPDATE;

            :NEW.IDHARVEST := v_next;
            UPDATE PLOT_HARVEST_SEQ
               SET NEXT_ID = v_next + 1
             WHERE IDPLOT = :NEW.IDPLOT;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                :NEW.IDHARVEST := 1;
                INSERT INTO PLOT_HARVEST_SEQ (IDPLOT, NEXT_ID)
                VALUES (:NEW.IDPLOT, 2);
        END;
    END IF;
END;
/
SHOW ERRORS TRIGGER HARVEST_BI_SET_ID;

--------------------------------------------------------
-- TRIGGER 2: Estado automático de la cosecha
--------------------------------------------------------
CREATE OR REPLACE TRIGGER HARVEST_BI_BU_STATUS
BEFORE INSERT OR UPDATE OF ENDDATE, STATUS_ID ON HARVEST
FOR EACH ROW
BEGIN
    IF INSERTING THEN
        IF :NEW.ENDDATE IS NOT NULL THEN
            :NEW.STATUS_ID := 2; -- FINALIZADO
        ELSIF :NEW.STATUS_ID IS NULL THEN
            :NEW.STATUS_ID := 1; -- EN_PROCESO
        END IF;
    END IF;

    IF :NEW.ENDDATE IS NOT NULL THEN
        :NEW.STATUS_ID := 2;
    END IF;

    IF :NEW.STATUS_ID = 2 AND :NEW.ENDDATE IS NULL THEN
        :NEW.ENDDATE := TRUNC(SYSDATE);
    END IF;
END;
/
SHOW ERRORS TRIGGER HARVEST_BI_BU_STATUS;

--------------------------------------------------------
-- TRIGGER 3: Validar una cosecha activa por lote
--------------------------------------------------------
CREATE OR REPLACE TRIGGER HARVEST_BI_VLD_ONE_ACTIVE
BEFORE INSERT OR UPDATE OF STATUS_ID, ENDDATE ON HARVEST
FOR EACH ROW
DECLARE
    v_exists NUMBER;
BEGIN
    IF (:NEW.STATUS_ID = 1 AND :NEW.ENDDATE IS NULL) THEN
        SELECT COUNT(*) INTO v_exists
          FROM HARVEST H
         WHERE H.IDPLOT = :NEW.IDPLOT
           AND H.STATUS_ID = 1
           AND H.ENDDATE IS NULL;

        IF v_exists > 0 THEN
            RAISE_APPLICATION_ERROR(
                -20051,
                'Solo se permite una cosecha activa por lote. IDPLOT='||:NEW.IDPLOT
            );
        END IF;
    END IF;
END;
/
SHOW ERRORS TRIGGER HARVEST_BI_VLD_ONE_ACTIVE;

--------------------------------------------------------
-- CONSULTAS DE PRUEBA 
-------------------------------------------------------- 
SELECT  
    P.PLOTNAME,
    H.IDHARVEST,
    H.STARTDATE,
    H.ENDDATE,
    S.STATUS_NAME AS HARVEST_STATUS,
    H.PRICEPERKILO
FROM HARVEST H
JOIN PLOT P ON P.IDPLOT = H.IDPLOT
JOIN HARVEST_STATUS_DOMAIN S ON S.STATUS_ID = H.STATUS_ID
ORDER BY P.IDPLOT, H.IDHARVEST;
